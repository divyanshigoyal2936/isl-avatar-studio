function buildHumanoidAvatar() {
      const avatarRoot = new THREE.Group();
      scene.add(avatarRoot);
      avatarRig.root = avatarRoot;

      // Realistic warm human skin tones (Cute female appearance)
      const skinColor = 0xdba178; // Warm soft Indian complexion
      const skinMat = new THREE.MeshStandardMaterial({
        color: skinColor,
        roughness: 0.58,
        metalness: 0.02
      });
      avatarRig.skinMat = skinMat;

      // Soft Blush / Lip shade
      const lipMat = new THREE.MeshStandardMaterial({
        color: 0xba5f58,
        roughness: 0.45,
        metalness: 0.0
      });

      // Elegant feminine attire (Chic pastel slate blazer & clean inner top)
      const outfitMat = new THREE.MeshStandardMaterial({
        color: 0x222a3d, // Deep indigo blazer
        roughness: 0.70,
        metalness: 0.08
      });

      const topMat = new THREE.MeshStandardMaterial({
        color: 0xf3f4f8, // Soft ivory inner top
        roughness: 0.60,
        metalness: 0.02
      });

      const accentMat = new THREE.MeshStandardMaterial({
        color: 0x1e293b,
        roughness: 0.70,
        metalness: 0.05
      });

      const hairMat = new THREE.MeshStandardMaterial({
        color: 0x16110e, // Rich dark brunette hair
        roughness: 0.82,
        metalness: 0.05
      });

      const eyeIrisMat = new THREE.MeshBasicMaterial({ color: 0x28180e });
      const scleraMat = new THREE.MeshBasicMaterial({ color: 0xfafafa });
      const browMat = new THREE.MeshStandardMaterial({ color: 0x16110e, roughness: 0.9 });

      // 1. Lower Body & Pelvis
      const pelvis = new THREE.Group();
      pelvis.position.y = 0.85;
      avatarRoot.add(pelvis);
      avatarRig.pelvis = pelvis;

      const legL = new THREE.Mesh(new THREE.CylinderGeometry(0.075, 0.055, 0.75, 18), outfitMat);
      legL.position.set(-0.10, -0.42, 0);
      pelvis.add(legL);

      const legR = new THREE.Mesh(new THREE.CylinderGeometry(0.075, 0.055, 0.75, 18), outfitMat);
      legR.position.set(0.10, -0.42, 0);
      pelvis.add(legR);

      // Elegant Shoes
      const shoeGeo = new THREE.BoxGeometry(0.090, 0.060, 0.19);
      const shoeMat = new THREE.MeshStandardMaterial({ color: 0x0f131a, roughness: 0.4 });
      const shoeL = new THREE.Mesh(shoeGeo, shoeMat);
      shoeL.position.set(-0.10, -0.80, 0.035);
      pelvis.add(shoeL);
      const shoeR = new THREE.Mesh(shoeGeo, shoeMat);
      shoeR.position.set(0.10, -0.80, 0.035);
      pelvis.add(shoeR);

      // 2. Spine & Torso (Soft feminine curves, not blocky)
      const spine = new THREE.Group();
      spine.position.y = 0.0;
      pelvis.add(spine);
      avatarRig.spine = spine;

      // Waist & Hips
      const waist = new THREE.Mesh(new THREE.CylinderGeometry(0.15, 0.165, 0.22, 24), outfitMat);
      waist.position.y = 0.11;
      spine.add(waist);

      // Upper Bust & Chest
      const bust = new THREE.Mesh(new THREE.CylinderGeometry(0.19, 0.15, 0.28, 24), outfitMat);
      bust.position.y = 0.35;
      spine.add(bust);

      // White inner top insert
      const topInsert = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.08, 0.16, 16), topMat);
      topInsert.position.set(0, 0.41, 0.085);
      spine.add(topInsert);

      // Natural subtle fabric button on blouse
      const button = new THREE.Mesh(new THREE.CylinderGeometry(0.008, 0.008, 0.004, 16), outfitMat);
      button.rotation.x = Math.PI / 2;
      button.position.set(0, 0.38, 0.155);
      spine.add(button);

      // Smooth Anatomical Clavicle / Shoulder Bridge
      const clavicleYoke = new THREE.Mesh(new THREE.CylinderGeometry(0.038, 0.038, 0.44, 20), outfitMat);
      clavicleYoke.rotation.z = Math.PI / 2;
      clavicleYoke.position.set(0, 0.44, 0.0);
      spine.add(clavicleYoke);

      // 3. Slender Neck & Expressive Female Head
      const neck = new THREE.Group();
      neck.position.set(0, 0.50, 0.015);
      spine.add(neck);
      avatarRig.neck = neck;

      const neckMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.048, 0.058, 0.12, 20), skinMat);
      neckMesh.position.y = 0.05;
      neck.add(neckMesh);

      // Natural neckline contour
      const neckShadow = new THREE.Mesh(new THREE.TorusGeometry(0.049, 0.003, 8, 24), outfitMat);
      neckShadow.rotation.x = Math.PI / 2;
      neckShadow.position.y = 0.03;
      neck.add(neckShadow);

      const head = new THREE.Group();
      head.position.set(0, 0.12, 0.01);
      neck.add(head);
      avatarRig.head = head;

      // Soft Humanoid Face Mesh (Cute almond shape)
      const headMesh = new THREE.Mesh(new THREE.SphereGeometry(0.128, 28, 28), skinMat);
      headMesh.scale.set(0.94, 1.22, 1.05);
      head.add(headMesh);

      // Soft rounded feminine chin
      const chinMesh = new THREE.Mesh(new THREE.ConeGeometry(0.065, 0.09, 20), skinMat);
      chinMesh.position.set(0, -0.085, 0.035);
      chinMesh.rotation.x = Math.PI;
      head.add(chinMesh);

      // Soft sculpted nose
      const noseGeo = new THREE.ConeGeometry(0.018, 0.054, 16);
      const nose = new THREE.Mesh(noseGeo, skinMat);
      nose.position.set(0, -0.012, 0.138);
      nose.rotation.x = Math.PI / 10;
      head.add(nose);

      // Expressive Lips (Natural smile & mouthing)
      const lipGeo = new THREE.BoxGeometry(0.046, 0.014, 0.020);
      const lips = new THREE.Mesh(lipGeo, lipMat);
      lips.position.set(0, -0.058, 0.134);
      head.add(lips);
      avatarRig.lips = lips;

      // Big, Expressive Almond Eyes with Blinking Eyelids
      function createEye(isLeft) {
        const eyeGroup = new THREE.Group();
        const xSign = isLeft ? -1 : 1;
        eyeGroup.position.set(xSign * 0.045, 0.022, 0.126);

        // Sclera
        const sclera = new THREE.Mesh(new THREE.SphereGeometry(0.020, 20, 20), scleraMat);
        sclera.scale.set(1.22, 0.85, 0.80);
        eyeGroup.add(sclera);

        // Warm Brown Iris
        const iris = new THREE.Mesh(new THREE.CircleGeometry(0.011, 20), eyeIrisMat);
        iris.position.set(0, 0, 0.017);
        eyeGroup.add(iris);

        // Cute Sparkling Highlights
        const glint1 = new THREE.Mesh(new THREE.CircleGeometry(0.0035, 12), new THREE.MeshBasicMaterial({ color: 0xffffff }));
        glint1.position.set(0.003, 0.003, 0.018);
        eyeGroup.add(glint1);

        const glint2 = new THREE.Mesh(new THREE.CircleGeometry(0.0018, 8), new THREE.MeshBasicMaterial({ color: 0xffffff }));
        glint2.position.set(-0.003, -0.003, 0.018);
        eyeGroup.add(glint2);

        // Eyelid for Smooth Blinking
        const lidGeo = new THREE.SphereGeometry(0.022, 20, 14, 0, Math.PI * 2, 0, Math.PI / 2);
        const lid = new THREE.Mesh(lidGeo, skinMat);
        lid.rotation.x = -Math.PI / 2;
        lid.position.set(0, 0.006, 0.003);
        eyeGroup.add(lid);

        return { eyeGroup, lid };
      }

      const leftEyeObj = createEye(true);
      const rightEyeObj = createEye(false);
      head.add(leftEyeObj.eyeGroup);
      head.add(rightEyeObj.eyeGroup);
      avatarRig.leftLid = leftEyeObj.lid;
      avatarRig.rightLid = rightEyeObj.lid;

      // Cute Dynamic Eyebrows (Physical motion for ISL markers)
      const browGeo = new THREE.BoxGeometry(0.042, 0.008, 0.010);
      const browL = new THREE.Mesh(browGeo, browMat);
      browL.position.set(-0.045, 0.054, 0.130);
      browL.rotation.z = -0.05;
      head.add(browL);
      avatarRig.browL = browL;

      const browR = new THREE.Mesh(browGeo, browMat);
      browR.position.set(0.045, 0.054, 0.130);
      browR.rotation.z = 0.05;
      head.add(browR);
      avatarRig.browR = browR;

      // Stylish Hair (Brunette with cute ponytail & bangs)
      const hairBase = new THREE.Mesh(new THREE.SphereGeometry(0.136, 24, 24), hairMat);
      hairBase.position.set(0, 0.038, -0.015);
      hairBase.scale.set(1.02, 1.20, 1.08);
      head.add(hairBase);

      // Cute High Ponytail
      const ponytail = new THREE.Mesh(new THREE.CylinderGeometry(0.032, 0.018, 0.24, 16), hairMat);
      ponytail.rotation.x = -0.45;
      ponytail.position.set(0, 0.06, -0.16);
      head.add(ponytail);

      // 4. Smooth Humanoid Arms (Natural shoulder & elbow curves, elbows stay DOWN!)
      function createArm(isLeft) {
        const xSign = isLeft ? -1 : 1;
        const armGroup = new THREE.Group();
        armGroup.position.set(xSign * 0.22, 0.44, 0.0);
        spine.add(armGroup);

        // Smooth Deltoid / Shoulder Contour (Natural human look)
        const shoulderJointMesh = new THREE.Mesh(new THREE.SphereGeometry(0.058, 24, 24), outfitMat);
        armGroup.add(shoulderJointMesh);

        // Shoulder Pivot
        const shoulder = new THREE.Group();
        armGroup.add(shoulder);

        // Upper Arm
        const upperArmMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.046, 0.040, 0.28, 20), skinMat);
        upperArmMesh.position.y = -0.14;
        shoulder.add(upperArmMesh);

        // Soft sleeve hem
        const sleeveHem = new THREE.Mesh(new THREE.CylinderGeometry(0.050, 0.048, 0.06, 20), outfitMat);
        sleeveHem.position.y = -0.03;
        shoulder.add(sleeveHem);

        // Smooth Elbow Pivot (Down by ribcage)
        const elbow = new THREE.Group();
        elbow.position.y = -0.28;
        shoulder.add(elbow);

        const elbowJointMesh = new THREE.Mesh(new THREE.SphereGeometry(0.044, 20, 20), skinMat);
        elbow.add(elbowJointMesh);

        // Forearm
        const forearmMesh = new THREE.Mesh(new THREE.CylinderGeometry(0.040, 0.034, 0.26, 20), skinMat);
        forearmMesh.position.y = -0.13;
        elbow.add(forearmMesh);

        // Wrist Joint
        const wrist = new THREE.Group();
        wrist.position.y = -0.26;
        elbow.add(wrist);

        const wristJointMesh = new THREE.Mesh(new THREE.SphereGeometry(0.032, 16, 16), skinMat);
        wrist.add(wristJointMesh);

        // Delicate feminine hand palm
        const palmGeo = new THREE.BoxGeometry(0.062, 0.076, 0.020);
        const palm = new THREE.Mesh(palmGeo, skinMat);
        palm.position.y = -0.038;
        wrist.add(palm);

        // 5 Fully Articulated Fingers
        const fingers = {};

        function buildFinger(name, posX, posY, baseLen) {
          const baseJoint = new THREE.Group();
          baseJoint.position.set(posX, posY, 0);
          wrist.add(baseJoint);

          // Knuckle
          const knuckle = new THREE.Mesh(new THREE.SphereGeometry(0.0065, 8, 8), skinMat);
          baseJoint.add(knuckle);

          // Phalanx 1
          const p1Mesh = new THREE.Mesh(new THREE.CylinderGeometry(0.0068, 0.0062, baseLen * 0.45, 8), skinMat);
          p1Mesh.position.y = -baseLen * 0.225;
          baseJoint.add(p1Mesh);

          // Mid Joint
          const midJoint = new THREE.Group();
          midJoint.position.y = -baseLen * 0.45;
          baseJoint.add(midJoint);

          // Phalanx 2
          const p2Mesh = new THREE.Mesh(new THREE.CylinderGeometry(0.0060, 0.0055, baseLen * 0.35, 8), skinMat);
          p2Mesh.position.y = -baseLen * 0.175;
          midJoint.add(p2Mesh);

          // Tip Joint
          const tipJoint = new THREE.Group();
          tipJoint.position.y = -baseLen * 0.35;
          midJoint.add(tipJoint);

          // Phalanx 3 / Fingertip
          const p3Mesh = new THREE.Mesh(new THREE.CylinderGeometry(0.0052, 0.0044, baseLen * 0.25, 8), skinMat);
          p3Mesh.position.y = -baseLen * 0.125;
          tipJoint.add(p3Mesh);

          return { base: baseJoint, mid: midJoint, tip: tipJoint };
        }

        // Thumb: Natural Opposable Angle
        const thumbX = xSign * 0.034;
        const thumbJoint = new THREE.Group();
        thumbJoint.position.set(thumbX, -0.025, 0.008);
        thumbJoint.rotation.z = xSign * 0.42;
        thumbJoint.rotation.y = xSign * 0.28;
        wrist.add(thumbJoint);

        const thumbP1 = new THREE.Mesh(new THREE.CylinderGeometry(0.0078, 0.0068, 0.030, 8), skinMat);
        thumbP1.position.y = -0.015;
        thumbJoint.add(thumbP1);

        const thumbTip = new THREE.Group();
        thumbTip.position.y = -0.030;
        thumbJoint.add(thumbTip);

        const thumbP2 = new THREE.Mesh(new THREE.CylinderGeometry(0.0065, 0.0054, 0.024, 8), skinMat);
        thumbP2.position.y = -0.012;
        thumbTip.add(thumbP2);

        fingers.thumb = { base: thumbJoint, mid: thumbTip, tip: thumbTip };

        // 4 Articulated Fingers
        const fingerSpacing = 0.015;
        fingers.index = buildFinger("index", xSign * (fingerSpacing * 1.5), -0.074, 0.068);
        fingers.middle = buildFinger("middle", xSign * (fingerSpacing * 0.5), -0.076, 0.075);
        fingers.ring = buildFinger("ring", -xSign * (fingerSpacing * 0.5), -0.074, 0.070);
        fingers.pinky = buildFinger("pinky", -xSign * (fingerSpacing * 1.5), -0.068, 0.056);

        return {
          shoulder,
          elbow,
          wrist,
          fingers
        };
      }

      avatarRig.leftArm = createArm(true);
      avatarRig.rightArm = createArm(false);

      applyPoseToAvatar(getRestingPose());
    }
